	\section{محاسبه گرادیان یک تابع اسکالر}
	
	تابع اسکالر زیر را در نظر بگیرید:
	\[
	S(x,y,z)=
	\left(x^2+y^2+z^2\right)^{-3/2}.
	\]
	گرادیان تابع \(S\) را در نقطه \((1,2,3)\) محاسبه کنید.
	همچنین اندازه گرادیان و کسینوس‌های جهتی آن را به دست آورید.
	
	\medskip

	\section{دیفرانسیل کامل یک تابع برداری}
	
	فرض کنید تابع برداری \(\mathbf{F}\) به مختصات مکانی
	و زمان وابسته باشد؛ یعنی
	\(\mathbf{F}=\mathbf{F}(\mathbf{r},t)\).
	نشان دهید که دیفرانسیل کامل آن به صورت زیر است:
	\[
	d\mathbf{F}
	=
	(d\mathbf{r}\cdot\nabla)\mathbf{F}
	+
	\frac{\partial\mathbf{F}}{\partial t}\,dt.
	\]
	
	\medskip

	\section{حرکت دایره‌ای و ضرب برداری}
	
	بردار مکان ذره‌ای که روی یک مسیر دایره‌ای با شعاع ثابت
	\(r\) و سرعت زاویه‌ای ثابت \(\omega\) حرکت می‌کند،
	به صورت زیر داده شده است:
	\[
	\mathbf{r}(t)
	=
	r\cos(\omega t)\mathbf{e}_x
	+
	r\sin(\omega t)\mathbf{e}_y.
	\]
	
	\begin{enumerate}[label=\textbf{\alph*)}]
		\item حاصل‌ضرب برداری
		\(\mathbf{r}\times\dot{\mathbf{r}}\) را محاسبه کنید.
		
		\item نشان دهید که معادله حرکت ذره به شکل زیر است:
		\[
		\ddot{\mathbf{r}}+\omega^2\mathbf{r}=0.
		\]
	\end{enumerate}
	
	\medskip

	\section{شرط کرل صفر برای یک میدان برداری}
	
	فرض کنید میدان برداری \(\mathbf{V}\) به طور کلی
	\lr{Irrotational} نباشد، اما تابع اسکالر \(g\) وجود داشته باشد،
	به طوری که میدان \(g\mathbf{V}\) شرط زیر را ارضا کند:
	\[
	\nabla\times(g\mathbf{V})=0.
	\]
	نشان دهید که:
	\[
	\mathbf{V}\cdot(\nabla\times\mathbf{V})=0.
	\]
	
	\medskip

	\section{اثبات اتحادهای آنالیز برداری}
	
	روابط زیر را ثابت کنید:
	
	\begin{enumerate}[label=\textbf{\alph*)}]
		
		\item
		\[
		\frac{d}{dt}(\mathbf{A}\cdot\mathbf{B})
		=
		\dot{\mathbf{A}}\cdot\mathbf{B}
		+
		\mathbf{A}\cdot\dot{\mathbf{B}}.
		\]
		
		\item
		\[
		\frac{d}{dt}(\mathbf{A}\times\mathbf{B})
		=
		\dot{\mathbf{A}}\times\mathbf{B}
		+
		\mathbf{A}\times\dot{\mathbf{B}}.
		\]
		
		\item
		\[
		\nabla\cdot(\mathbf{a}\times\mathbf{b})
		=
		\mathbf{b}\cdot(\nabla\times\mathbf{a})
		-
		\mathbf{a}\cdot(\nabla\times\mathbf{b}).
		\]
		
		\item
		\[
		\begin{aligned}
			\nabla\times(\mathbf{A}\times\mathbf{B})
			={}&
			(\mathbf{B}\cdot\nabla)\mathbf{A}
			-
			(\mathbf{A}\cdot\nabla)\mathbf{B}\\
			&-\mathbf{B}(\nabla\cdot\mathbf{A})
			+\mathbf{A}(\nabla\cdot\mathbf{B}).
		\end{aligned}
		\]
		
		\item
		\[
		\begin{aligned}
			\nabla(\mathbf{A}\cdot\mathbf{B})
			={}&
			(\mathbf{A}\times\nabla)\times\mathbf{B}\\
			&+(\mathbf{B}\times\nabla)\times\mathbf{A}\\
			&+\mathbf{A}(\nabla\cdot\mathbf{B})
			+\mathbf{B}(\nabla\cdot\mathbf{A}).
		\end{aligned}
		\]
		
		\item
		\[
		\mathbf{A}\times(\nabla\times\mathbf{A})
		=
		\frac{1}{2}\nabla(\mathbf{A}^2)
		-
		(\mathbf{A}\cdot\nabla)\mathbf{A}.
		\]
		
		\item اگر \(\mathbf{A}\) و \(\mathbf{B}\) بردارهای ثابت باشند،
		نشان دهید که:
		\[
		\nabla\left[
		\mathbf{A}\cdot(\mathbf{B}\times\mathbf{r})
		\right]
		=
		\mathbf{A}\times\mathbf{B}.
		\]
		
		\item با بسط مستقیم در مختصات کارتزین، نشان دهید که:
		\[
		\nabla\times(\nabla\times\mathbf{V})
		=
		\nabla(\nabla\cdot\mathbf{V})
		-
		\nabla^2\mathbf{V}.
		\]
		
		\item
		\[
		\nabla\times(\phi\nabla\phi)=0.
		\]
		
		\item اگر
		\[
		\nabla\times\mathbf{F}
		=
		\nabla\times\mathbf{G},
		\]
		نشان دهید که در یک ناحیه مناسب، اختلاف دو میدان را
		می‌توان به صورت گرادیان یک تابع اسکالر نوشت:
		\[
		\mathbf{F}-\mathbf{G}=\nabla\phi.
		\]
		
		\item نشان دهید که:
		\[
		\begin{aligned}
			&\nabla\times
			[(\mathbf{v}\cdot\nabla)\mathbf{v}]\\
			&\qquad=
			-\nabla\times
			[\mathbf{v}\times(\nabla\times\mathbf{v})].
		\end{aligned}
		\]
		
	\end{enumerate}
	
	\medskip

	\section{حل معادله رسانش گرما}
	
	معادله رسانش گرما به صورت زیر داده شده است:
	\[
	\nabla^2\psi=k|\nabla\phi|^2,
	\]
	که در آن \(\phi\) معادله لاپلاس را ارضا می‌کند:
	\[
	\nabla^2\phi=0.
	\]
	نشان دهید که تابع زیر جوابی برای معادله داده شده است:
	\[
	\psi=\frac{k}{2}\phi^2.
	\]
	
	\medskip

	\section{بازنویسی معادلات ماکسول با استفاده از ماتریس‌ها}
	
	ماتریس‌های زیر را در نظر بگیرید:
	\[
	M_x=
	\begin{pmatrix}
		0&0&0\\
		0&0&-i\\
		0&i&0
	\end{pmatrix},
	\qquad
	M_y=
	\begin{pmatrix}
		0&0&i\\
		0&0&0\\
		-i&0&0
	\end{pmatrix},
	\]
	\[
	M_z=
	\begin{pmatrix}
		0&-i&0\\
		i&0&0\\
		0&0&0
	\end{pmatrix}.
	\]
	
	با تعریف مولفه‌های بردار ستونی \(\boldsymbol{\psi}\) به صورت
	\[
	\psi_j=B_j-\frac{iE_j}{c},
	\qquad j=x,y,z,
	\]
	نشان دهید که معادله ماتریسی
	\[
	\left(
	\mathbf{M}\cdot\nabla
	+
	\frac{1}{c}I_3\frac{\partial}{\partial t}
	\right)
	\boldsymbol{\psi}=0
	\]
	معادلات ماکسول در خلأ را بازتولید می‌کند.
	در این رابطه \(I_3\) ماتریس همانی مرتبه سه است.
	
	\medskip

	\section{اثبات اتحاد ماتریس‌های پائولی}
	
	با استفاده از ماتریس‌های پائولی زیر، رابطه داده شده را ثابت کنید:
	\[
	\sigma_1=
	\begin{pmatrix}
		0&1\\
		1&0
	\end{pmatrix},
	\qquad
	\sigma_2=
	\begin{pmatrix}
		0&-i\\
		i&0
	\end{pmatrix},
	\qquad
	\sigma_3=
	\begin{pmatrix}
		1&0\\
		0&-1
	\end{pmatrix}.
	\]
	
	بردار ماتریسی پائولی را به صورت
	\[
	\boldsymbol{\sigma}
	=
	\mathbf{e}_x\sigma_1
	+
	\mathbf{e}_y\sigma_2
	+
	\mathbf{e}_z\sigma_3
	\]
	تعریف کنید. نشان دهید که:
	\[
	(\boldsymbol{\sigma}\cdot\mathbf{a})
	(\boldsymbol{\sigma}\cdot\mathbf{b})
	=
	(\mathbf{a}\cdot\mathbf{b})I_2
	+
	i\boldsymbol{\sigma}\cdot
	(\mathbf{a}\times\mathbf{b}),
	\]
	که در آن \(\mathbf{a}\) و \(\mathbf{b}\) بردارهای معمولی و
	\(I_2\) ماتریس همانی مرتبه دو است.

	\section{عملگر تکانه زاویه‌ای و روابط جابه‌جایی}
	
	با استفاده از تعریف عملگر تکانه زاویه‌ای
	\[
	\mathbf{L}=\mathbf{r}\times\mathbf{p},
	\qquad
	\mathbf{p}=-i\nabla,
	\]
	و با فرض \(\hbar=1\)، مولفه‌های کارتزین عملگر تکانه زاویه‌ای
	را به دست آورید:
	\[
	L_x=-i\left(
	y\frac{\partial}{\partial z}
	-z\frac{\partial}{\partial y}
	\right),
	\]
	\[
	L_y=-i\left(
	z\frac{\partial}{\partial x}
	-x\frac{\partial}{\partial z}
	\right),
	\]
	\[
	L_z=-i\left(
	x\frac{\partial}{\partial y}
	-y\frac{\partial}{\partial x}
	\right).
	\]
	
	سپس نشان دهید که روابط زیر برقرارند:
	\[
	[L_x,L_y]=iL_z,
	\qquad
	\mathbf{L}\times\mathbf{L}=i\mathbf{L}.
	\]
	
	در ادامه، اگر بردارهای \(\mathbf{a}\) و \(\mathbf{b}\) با یکدیگر
	و با مولفه‌های عملگر \(\mathbf{L}\) جابه‌جایی‌پذیر باشند،
	ثابت کنید که:
	\[
	[\mathbf{a}\cdot\mathbf{L},
	\mathbf{b}\cdot\mathbf{L}]
	=
	i(\mathbf{a}\times\mathbf{b})\cdot\mathbf{L}.
	\]
