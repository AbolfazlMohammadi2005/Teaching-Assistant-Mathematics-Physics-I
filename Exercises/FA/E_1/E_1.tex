	\section{دترمینان ماتریس‌ها}
	
	دترمینان ماتریس‌های زیر را محاسبه کنید:
	
	\[
	A=
	\begin{pmatrix}
		3&2\\
		1&4
	\end{pmatrix}
	\]
	
	و
	
	\[
	B=
	\begin{pmatrix}
		2&1&3\\
		0&4&2\\
		1&2&1
	\end{pmatrix}.
	\]

	\section{وارون ماتریس‌ها}
	
	وارون ماتریس‌های زیر را در صورت وجود محاسبه کنید. پیش از محاسبه‌ی وارون، شرط وارون‌پذیری ماتریس را بررسی کنید:
	
	\[
	A=
	\begin{pmatrix}
		3&2\\
		1&4
	\end{pmatrix}

	و
	
	\[
	B=
	\begin{pmatrix}

	\section{دوران در فضای سه‌بعدی}
	
	نقطه‌ای با مختصات
	
	\[
	P=(X,Y,Z)
	\]
	
	را در نظر بگیرید. مختصات نقطه پس از دوران را با
	
	\[
	P'=(X',Y',Z')
	\]
	
	نشان می‌دهیم.
	
	\subsection*{الف)}
	
	روابط تبدیل مختصات $X'$، $Y'$ و $Z'$ را برای دوران نقطه $P$ به اندازه‌ی زاویه‌ی $\theta$ حول محور $Z$ به دست آورید و تبدیل را به صورت یک رابطه‌ی ماتریسی بنویسید.
	
	\subsection*{ب)}
	
	روابط تبدیل مختصات $X'$، $Y'$ و $Z'$ را برای دوران به اندازه‌ی زاویه‌ی $\theta$ حول محور $Y$ به دست آورید و ماتریس دوران متناظر را بنویسید.
	
	\subsection*{پ)}
	
	روابط تبدیل مختصات $X'$، $Y'$ و $Z'$ را برای دوران به اندازه‌ی زاویه‌ی $\theta$ حول محور $X$ به دست آورید و ماتریس دوران متناظر را بنویسید.
	
	در هر سه حالت، هر سه رابطه‌ی مختصات را به‌طور کامل استخراج کرده و سپس آن‌ها را در قالب یک ماتریس $3\times3$ نمایش دهید.

	\section{ویژگی‌های ماتریس‌های دوران}
	
	با استفاده از سه ماتریس دوران به‌دست‌آمده در سؤال قبل، ویژگی‌های زیر را بررسی و اثبات کنید:
	
	\subsection*{الف)}
	
	نشان دهید:
	
	\[
	\det(R)=1
	\]
	
	\subsection*{ب)}
	
	نشان دهید:
	
	\[
	R^TR=I
	\]
	
	\subsection*{پ)}
	
	نشان دهید:
	
	\[
	R^{-1}=R^T
	\]
	
	\subsection*{ت)}
	
	نشان دهید:
	
	\[
	R^{-1}(\theta)=R(-\theta)
	\]
	
	\subsection*{ث)}
	
	نشان دهید حاصل‌ضرب دو ماتریس دوران حول یک محور یکسان، خود یک ماتریس دوران حول همان محور است و زاویه‌ی دوران حاصل برابر مجموع دو زاویه خواهد بود.

	\section{انتقال در فضای سه‌بعدی}
	
	نقطه‌ای با مختصات
	
	\[
	P=
	\begin{pmatrix}
		X\\
		Y\\
		Z
	\end{pmatrix}
	\]
	
	را در نظر بگیرید که تحت انتقالی با بردار
	
	\[
	\mathbf{a}=
	\begin{pmatrix}
		a\\
		b\\
		c
	\end{pmatrix}
	\]
	
	قرار می‌گیرد.
	
	\subsection*{الف)}
	
	مختصات نقطه‌ی جدید $P'$ را بر حسب $X,Y,Z,a,b,c$ به دست آورید.
	
	\subsection*{ب)}
	
	تبدیل انتقال را در قالب یک ماتریس $4\times4$ در مختصات همگن نمایش دهید.
	
	\subsection*{پ)}
	
	ماتریس انتقال معکوس را به دست آورید و نشان دهید که اعمال آن بر نقطه‌ی $P'$، نقطه‌ی اولیه $P$ را بازمی‌گرداند.
	
	\subsection*{ت)}
	
	نشان دهید:
	
	\[
	T^{-1}T=TT^{-1}=I.
	\]

	\section{ترکیب دوران و انتقال}
	
	یک نقطه $P$ در فضای سه‌بعدی را در نظر بگیرید. فرض کنید $R$ یک ماتریس دوران و $T$ یک ماتریس انتقال باشد.
	
	\subsection*{الف)}
	
	نشان دهید اگر ابتدا نقطه را دوران داده و سپس منتقل کنیم، تبدیل کلی به صورت
	
	\[
	P'=RP+\mathbf{t}
	\]
	
	خواهد بود.
	
	در مختصات همگن، ماتریس تبدیل متناظر را به صورت
	
	\[
	T=
	\begin{pmatrix}
		R&\mathbf{t}\\
		0&1
	\end{pmatrix}
	\]
	
	بنویسید.
	
	\subsection*{ب)}
	
	اگر ابتدا نقطه را منتقل کرده و سپس دوران دهیم، نشان دهید:
	
	\[
	P''=R(P+\mathbf{t})
	\]
	
	و تبدیل متناظر را به صورت
	
	\[
	T_R=
	\begin{pmatrix}
		R&R\mathbf{t}\\
		0&1
	\end{pmatrix}
	\]
	
	بنویسید.
	
	\subsection*{پ)}
	
	با مقایسه‌ی دو تبدیل بالا نشان دهید که در حالت کلی:
	
	\[
	\boxed{
		TT_R\neq T_RT
	}
	\]
	
	و بنابراین نشان دهید دوران و انتقال در حالت کلی جابه‌جایی‌پذیر نیستند.
	
	\subsection*{ت)}
	
	این نتیجه را با استفاده از ضرب صریح ماتریس‌های $4\times4$ در مختصات همگن اثبات کنید.

	\section{دلتای کرونکر و نمادگذاری اندیسی}
	
	با استفاده از قرارداد جمع اینشتین و خواص دلتای کرونکر
	
	\[
	\delta_{ij}=
	\begin{cases}
		1&i=j\\
		0&i\neq j
	\end{cases}
	\]
	
	عبارت زیر را در نظر بگیرید:
	
	\[
	T_{ij}
	=
	\delta_{ij}A_kB_k
	-
	A_iB_j
	+
	\delta_{ik}A_kB_j
	-
	\delta_{jk}A_iB_k.
	\]
	
	\subsection*{الف)}
	
	عبارت $T_{ij}$ را با استفاده از دلتای کرونکر تا حد ممکن ساده کنید.
	
	\subsection*{ب)}
	
	نشان دهید که $T_{ij}$ را می‌توان تنها بر حسب $A_i$، $B_i$، حاصل‌ضرب داخلی $\mathbf{A}\cdot\mathbf{B}$ و $\delta_{ij}$ نوشت.
	
	\subsection*{پ)}
	
	با محاسبه‌ی
	
	\[
	T_{ii}
	\]
	
	ردیابی $T_{ij}$ را به دست آورید.
	
	\subsection*{ت)}
	
	بررسی کنید که آیا $T_{ij}$ نسبت به جابه‌جایی دو شاخص متقارن است یا خیر؛ یعنی مشخص کنید آیا
	
	\[
	T_{ij}=T_{ji}
	\]
	
	برقرار است یا نه.
	
	\subsection*{ث)}
	
	اگر
	
	\[
	A_iA_i=1,
	\qquad
	B_iB_i=1,
	\qquad
	A_iB_i=0,
	\]
	
	باشد، عبارت نهایی $T_{ij}$ و مقدار $T_{ii}$ را به دست آورید.

	\section{سؤال امتیازی: دوران حول یک محور دلخواه}
	
	بردار $\mathbf{A}$ را در نظر بگیرید که حول محور واحد
	
	\[
	\hat{\mathbf{n}}
	\]
	
	به اندازه‌ی زاویه‌ی $\phi$ دوران داده می‌شود، به‌طوری‌که:
	
	\[
	|\hat{\mathbf{n}}|=1.
	\]
	
	\subsection*{الف)}
	
	با تجزیه‌ی بردار به مؤلفه‌های موازی و عمود بر محور دوران، رابطه‌ی بردار دوران‌یافته $\mathbf{A}'$ را به دست آورید.
	
	\subsection*{ب)}
	
	نشان دهید:
	
	\[
	\boxed{
		\mathbf{A}'
		=
		\mathbf{A}\cos\phi
		+
		(\hat{\mathbf{n}}\times\mathbf{A})\sin\phi
		+
		\hat{\mathbf{n}}(\hat{\mathbf{n}}\cdot\mathbf{A})(1-\cos\phi)
	}
	\]
	
	\subsection*{پ)}
	
	نشان دهید رابطه‌ی بالا را می‌توان به صورت
	
	\[
	\boxed{
		\mathbf{A}'
		=
		\mathbf{A}
		+
		(\hat{\mathbf{n}}\times\mathbf{A})\sin\phi
		+
		\hat{\mathbf{n}}\times
		(\hat{\mathbf{n}}\times\mathbf{A})(1-\cos\phi)
	}
	\]
	
	نوشت.
	
	\subsection*{ت)}
	
	نشان دهید دوران اندازه‌ی بردار را تغییر نمی‌دهد:
	
	\[
	\boxed{
		|\mathbf{A}'|=|\mathbf{A}|
	}
	\]
	
	\subsection*{ث)}
	
	عملگر دوران $D_{\hat{\mathbf{n}}}(\phi)$ را به‌گونه‌ای به دست آورید که:
	
	\[
	\boxed{
		\mathbf{A}'
		=
		D_{\hat{\mathbf{n}}}(\phi)\mathbf{A}
	}
	\]
	
	باشد.

	\section{سؤال امتیازی: گروه‌های متعامد و دوران}
	
	گروه‌های متعامد $O(n)$ و گروه‌های متعامد ویژه $SO(n)$ را در نظر بگیرید:
	
	\[
	O(n)=
	\{R\in M_n(\mathbb{R})\mid R^TR=I\}
	\]
	
	و
	
	\[
	SO(n)=
	\{R\in O(n)\mid\det R=1\}.
	\]
	
	\subsection*{الف)}
	
	نشان دهید برای هر ماتریس $R\in O(n)$:
	
	\[
	\det(R)=\pm1.
	\]
	
	\subsection*{ب)}
	
	ماتریس‌های عمومی گروه‌های
	
	\[
	O(1),\qquad O(2),\qquad O(3)
	\]
	
	و همچنین
	
	\[
	SO(1),\qquad SO(2),\qquad SO(3)
	\]
	
	را مشخص کنید.
	
	\subsection*{پ)}
	
	نشان دهید اعضای $SO(2)$ به صورت
	
	\[
	R(\theta)=
	\begin{pmatrix}
		\cos\theta&-\sin\theta\\
		\sin\theta&\cos\theta
	\end{pmatrix}
	\]
	
	نوشته می‌شوند و ثابت کنید:
	
	\[
	R(\theta_1)R(\theta_2)
	=
	R(\theta_1+\theta_2).
	\]
	
	\subsection*{ت)}
	
	با استفاده از ماتریس‌های دوران سه‌بعدی سؤال سوم، نشان دهید ماتریس‌های دوران حول محورهای $x$، $y$ و $z$ عضو
	
	\[
	SO(3)
	\]
	
	هستند.
	
	\subsection*{ث)}
	
	تفاوت $O(3)$ و $SO(3)$ را از نظر دترمینان و نوع تبدیل هندسی توضیح دهید.
	
	\subsection*{ج)}
	
	نشان دهید $SO(3)$ تحت عمل ضرب ماتریسی یک گروه تشکیل می‌دهد؛ یعنی بسته بودن، عضو واحد و وجود عضو معکوس را بررسی کنید.
	
	\subsection*{چ)}
	
	مشخص کنید آیا مجموعه‌ی تمام ماتریس‌های دوران حول محور $x$، یک زیرگروه از $SO(3)$ تشکیل می‌دهد یا خیر. استدلال خود را ارائه دهید.
